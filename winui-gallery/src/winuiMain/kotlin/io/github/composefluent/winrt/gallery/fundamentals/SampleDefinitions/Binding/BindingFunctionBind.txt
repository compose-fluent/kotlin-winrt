--- header
Using a function in x:Bind
--- xaml
<DatePicker x:Name="DatePickerControl" Header="Select a date" />
<TextBlock Text="{x:Bind FormatDate(DatePickerControl.SelectedDate), Mode=OneWay}" />
--- kotlin
fun FormatDate(date: kotlin.time.Instant?): String = date?.let {
    "Selected date is: ${windows.globalization.datetimeformatting.DateTimeFormatter("dayofweek month day year").format(it)}"
} ?: "No date selected"
